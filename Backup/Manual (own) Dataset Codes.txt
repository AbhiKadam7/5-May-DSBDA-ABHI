# Create dataset

    'Age':[22,25,None,30],

🔥 Q2: Missing + Outliers + Transformation
import numpy as np

# Fill missing
df.fillna(df.mean(numeric_only=True), inplace=True)

# Remove outliers (IQR)
Q1 = df.quantile(0.25)
Q3 = df.quantile(0.75)
IQR = Q3 - Q1

df = df[~((df < (Q1-1.5*IQR)) | (df > (Q3+1.5*IQR))).any(axis=1)]

# Transformation
df['Salary'] = np.log(df['Salary'])

print(df)



🔥 Q3: Group Statistics

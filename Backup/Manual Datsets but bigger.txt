🔥 Q2: Missing + Outliers + Transformation
# Q2: Data Cleaning + Transformation

import pandas as pd
import numpy as np

data = {
    'Marks':[85,90,88,120,95,78,82,91,87,93,89,110,84],
    'StudyHours':[5,6,5,10,7,4,5,6,5,7,6,9,5]
}

df = pd.DataFrame(data)

# Fill missing (if any)
df.fillna(df.mean(numeric_only=True), inplace=True)

# Remove outliers
Q1 = df.quantile(0.25)
Q3 = df.quantile(0.75)
IQR = Q3 - Q1
df = df[~((df < (Q1-1.5*IQR)) | (df > (Q3+1.5*IQR))).any(axis=1)]

# Transformation
df['Marks'] = np.log(df['Marks'])

print(df)

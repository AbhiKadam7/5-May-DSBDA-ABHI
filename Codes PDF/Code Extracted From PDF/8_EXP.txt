import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Load dataset
titanic = sns.load_dataset('titanic')

# Show full dataset
print(titanic)

# Show first 5 rows
print("\nFirst 5 Rows:\n", titanic.head())

# Show last 5 rows
print("\nLast 5 Rows:\n", titanic.tail())

# Shape of dataset
print("\nShape:", titanic.shape)

# Info
print("\nInfo:\n")
print(titanic.info())

# Histogram for age
sns.displot(titanic['age'], bins=10)

# Histogram for fare
sns.displot(titanic['fare'], bins=10)

# Countplot for survived
sns.countplot(x='survived', data=titanic)

# Countplot for sex
sns.countplot(x='sex', data=titanic)

# Countplot for pclass
sns.countplot(x='pclass', data=titanic)

# Show dataset again
print("\nDataset:\n", titanic)

# Show plots
plt.show()

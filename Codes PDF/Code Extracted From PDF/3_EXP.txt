import pandas as pd
from sklearn import preprocessing

# Load dataset
df = pd.read_csv("C:/Users/Dell/Mall_Customers.csv")

# Show full dataset
print(df)

# Show first 5 rows
print("\nFirst 5 Rows:\n", df.head())

# Show last 5 rows
print("\nLast 5 Rows:\n", df.tail())

# Shape of dataset
print("\nShape:", df.shape)

# Info
print("\nInfo:\n")
print(df.info())

# Statistics
print("\nStatistics:\n", df.describe())

# Minimum values
print("\nMin Values:\n", df.min())

# Maximum values
print("\nMax Values:\n", df.max())

# Groupby mean
print("\nMean Age by Genre:\n", df.groupby(["Genre"])["Age"].mean())

# Groupby median
print("\nMedian Age by Genre:\n", df.groupby("Genre")["Age"].median())

# Groupby std
print("\nStd Age by Genre:\n", df.groupby(df['Genre'])['Age'].std())

# Mean income by Genre
print("\nMean Income by Genre:\n", df.groupby('Genre')['Annual Income (k$)'].mean())

# Groupby mean (all columns)
print("\nGroupby Mean:\n", df.groupby('Genre').mean())

# Groupby median (all columns)
print("\nGroupby Median:\n", df.groupby(['Genre']).median())

# Groupby std (all columns)
print("\nGroupby Std:\n", df.groupby(['Genre']).std())

# Show dataset again
print("\nDataset:\n", df)

# Drop Genre column
x = df.drop(axis=1, columns=['Genre'])

print("\nAfter Dropping Genre:\n", x)

# Show dataset again
print("\nDataset:\n", df)

# One Hot Encoding
enc = preprocessing.OneHotEncoder()
enc_df = pd.DataFrame(enc.fit_transform(df[['Genre']]).toarray())

print("\nEncoded Data:\n", enc_df)

# Join encoded data
df_encode = x.join(enc_df)

print("\nFinal Encoded Dataset:\n", df_encode)
